    // =========================================================================
// 10-components/src/app/doctor-card/doctor-card.ts - eine KIND-Komponente
// -------------------------------------------------------------------------
// Sie erhält ihre Daten von ihrem Elternelement über INPUTS, deklariert
// mit input(): Jeder Input ist ein schreibgeschütztes SIGNAL, das dem
// vom Elternelement übergebenen Wert folgt. Ihr Typ wird zur Kompilierzeit geprüft:
// <app-doctor-card [doctor]="42"> würde abgelehnt.
//
// Goldene Regel: Eine Komponente ändert ihre Inputs NIEMALS. Die Daten
// fließen vom Elternelement zum Kind; um eine Änderung anzufordern,
// löst das Kind ein Ereignis aus (siehe Beispiel 11).
// =========================================================================

import { booleanAttribute, Component, computed, input } from '@angular/core';
import { Doctor } from '../doctor';

@Component({
  // der Selektor: das Tag, das diese Komponente in ihrem Elternelement repräsentiert
  selector: 'app-doctor-card',
  templateUrl: './doctor-card.html',
  styleUrl: './doctor-card.css',
})
export class DoctorCard {
  // input.required: das Elternelement MUSS diesen Input angeben
  readonly doctor = input.required<Doctor>();
  readonly rank = input.required<number>();
  // input(Standardwert): ein optionaler Input. booleanAttribute
  // erlaubt es, einfach <app-doctor-card compact> für true zu schreiben
  readonly compact = input(false, { transform: booleanAttribute });

  // ein computed auf Basis eines Inputs: wird neu berechnet, wenn das Elternelement ihn ändert
  protected readonly initials = computed(() => `${this.doctor().firstName[0]}${this.doctor().lastName[0]}`);
}
    • Zeile 19: der Selektor: Die Elternkomponente schreibt [<app-doctor-card>];
    • Zeilen 25–29: die Inputs der Komponente ([inputs]) – die Daten, die ihr von ihrer Elternkomponente übergeben werden. [input()] erzeugt ein schreibgeschütztes Signal, das den von der Elternkomponente übergebenen Wert verfolgt:
      • Zeilen 25–26: [input.required<T>()]: ein erforderlicher Input. Fehlt [[doctor]] im Template der Elternkomponente, führt das zu einem Kompilierungsfehler;
      • Zeile 29: [input(default value)]: ein optionaler Input. Die Option [transform] wandelt den empfangenen Wert um: [booleanAttribute] wandelt das Vorhandensein des Attributs ([<app-doctor-card compact>], ohne Wert) in ein [true] um, wie beim HTML-Attribut [disabled];
    • Zeile 32: ein [computed], das auf einem Input beruht: wird neu berechnet, wenn die Elternkomponente den Input ändert.

    Die Inputs sind nicht [protected]: Sie gehören zur öffentlichen Schnittstelle der Komponente, die ihre Elternkomponente verwendet.

    Goldene Regel: Eine Komponente ändert ihre Inputs nie (sie sind ohnehin schreibgeschützt). Die Daten fließen von der Eltern- zur Kindkomponente; um eine Änderung anzufordern, sendet die Kindkomponente ein Ereignis (Beispiel 11).

// 10-components/src/app/app.ts - die ELTERN-Komponente
// -------------------------------------------------------------------------
// Sie importiert die Komponente DoctorCard (Array imports) und verwendet sie als
// neues HTML-Tag, <app-doctor-card>, so oft wie
// nötig: Eine Komponente ist ein WIEDERVERWENDBARER Teil der Oberfläche mit
// eigenen Daten, eigenem Template und eigenem Stil.

import { Component, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';

import { Doctor } from './doctor';

@Component({
  selector: 'app-root',
  imports: [FormsModule, DoctorCard],
  templateUrl: './app.html',
})
export class App {
  protected readonly doctors = signal<Doctor[]>([
    { id: 1, title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER', specialty: 'general practitioner', available: true },
    { id: 2, title: 'Mr', firstName: 'Paul', lastName: 'TATOU', specialty: 'pediatrician', available: false },

    this.doctors.update((list) => list.map((m) => ({ ...m, available: !m.available })));
  }
}

    // =========================================================================
// 11-outputs-model/src/app/counter/counter.ts
// -------------------------------------------------------------------------
// Ein Kind, das sein Elternelement BENACHRICHTIGT: output() deklariert ein Ereignis, das die
// Komponente auslösen kann, mit dem Typ seiner Daten. Das Elternelement hört darauf
// wie auf ein HTML-Ereignis: <app-counter (incremented)="..." (limit)="...">.
// =========================================================================

import { Component, input, output, signal } from '@angular/core';

@Component({
  selector: 'app-counter',
  template: `
    <div class="d-flex align-items-center gap-2 mb-2">
      <span class="w-25">{{ label() }}</span>
      <button class="btn btn-sm btn-outline-primary" (click)="plus()">+1</button>
      <span class="badge text-bg-secondary">{{ value() }} / {{ max() }}</span>
    </div>
  `,
})
export class Counter {
  readonly label = input.required<string>();
  readonly max = input.required<number>();

  // zwei Ereignisse: incremented (mit dem neuen Wert) und limit (ohne Daten)
  readonly incremented = output<number>();
  readonly limit = output<void>();

  protected readonly value = signal(0);

  protected plus(): void {
    if (this.value() >= this.max()) {
      this.limit.emit();
      return;
    }
    this.value.update((v) => v + 1);
    this.incremented.emit(this.value());
  }
}
    • Zeilen 13–19: das Template der Komponente, im Dekorator geschrieben (Eigenschaft [template], in Backticks eingeschlossen);
    • Zeilen 26–27: [output<T>()] deklariert einen Output: ein Ereignis, das die Komponente senden kann, samt dem Typ seiner Daten. [limit] hat keine Daten ([void]);
    • Zeilen 33 und 37: [emit(data)] sendet das Ereignis; die Elternkomponente erhält die Daten.

    Vermeiden Sie es, einen Output nach einem Browser-Ereignis zu benennen ([change], [click] ...): Browser-Ereignisse, die von den Elementen der Komponente ausgelöst werden, propagieren bis zu ihrem Host-Element, und die Elternkomponente würde sie ebenfalls empfangen.

    // =========================================================================
// 11-outputs-model/src/app/text-field/text-field.ts
// -------------------------------------------------------------------------
// Ein "selbstgeschriebenes" Formularfeld (Label + Feld + Hilfe + Zeichenzähler),
// das wie ngModel ein ZWEI-Wege-Binding akzeptiert:
//     <app-text-field [(value)]="lastName" label="Last name" [max]="20" />
//
// model() erstellt ein BESCHREIBBARES Signal, das mit dem Elternelement verknüpft ist: LESEN liefert den
// Wert des Elternelements, SCHREIBEN (set) löst das Ereignis valueChange aus, das die
// Syntax [( )] in eine Aktualisierung des Signals des Elternelements umsetzt. Die Daten
// bleiben so tatsächlich Eigentum des Elternelements.
// =========================================================================

import { Component, input, model } from '@angular/core';

// ein von allen Instanzen geteilter Zähler: eine eindeutige Kennung pro
// Feld, um <label for> und <input id> zu verknüpfen
let nextNumber = 0;

@Component({
  selector: 'app-text-field',
  template: `
    <div class="mb-3">
      <label class="form-label" [for]="id">{{ label() }}</label>
      <!-- [value]: der Wert des Elternelements; (input): jeder Tastendruck ersetzt ihn -->
      <input class="form-control" [id]="id" [value]="value()" [attr.maxlength]="max()"
             [class.is-invalid]="value().length === max()" (input)="onInput($event)">
      <div class="form-text">{{ help() }} ({{ value().length }} / {{ max() }})</div>
    </div>
  `,
})
export class TextField {
  readonly value = model.required<string>();
  readonly label = input.required<string>();
  readonly max = input.required<number>();
  readonly help = input('');

  protected readonly id = `text-field-${++nextNumber}`;

  protected onInput(event: Event): void {
    this.value.set((event.target as HTMLInputElement).value);
  }
}
    • Zeile 33: [model()] erzeugt ein beschreibbares Signal, das mit der Elternkomponente verknüpft ist. Beim LESEN liefert es den Wert der Elternkomponente; beim SCHREIBEN ([set], Zeile 41) ändert es den Wert UND sendet das Ereignis [valueChange] (der Modellname gefolgt von [Change]). Bei der Elternkomponente, die [[(value)]="signal"] schreibt, wird somit ihr Signal aktualisiert;
    • Zeile 18: eine Modulvariable (außerhalb der Klasse) wird von allen Instanzen der Komponente gemeinsam genutzt: ein Zähler, der jedem Feld einen eindeutigen Bezeichner zuweist (Zeile 38), um [<label for>] und [<input id>] zu verknüpfen;
    • Zeilen 26–27: [[value]] zeigt den Wert der Elternkomponente an; [(input)] ersetzt ihn bei jedem Tastendruck (Zeile 41). [[attr.maxlength]] ist ein Attribut-Binding an HTML (das Attribut [maxlength]), kein Property Binding.

// 11-outputs-model/src/app/app.ts - das Elternelement hört auf seine Kinder
// -------------------------------------------------------------------------
// Zwei Wege, wie ein Kind Informationen nach oben gibt:
//   1. ein Ereignis auslösen (output), auf das das Elternelement mit ( ) hört;
//   2. ein Zwei-Wege-Binding [( )] auf einem model(): eine Kurzform
//      für den Input [value] + das Ereignis (valueChange).
// =========================================================================

import { Component, computed, signal } from '@angular/core';
import { Counter } from './counter/counter';
import { TextField } from './text-field/text-field';

@Component({
  selector: 'app-root',
  imports: [Counter, TextField],
  templateUrl: './app.html',
})
export class App {
  protected readonly total = signal(0);
  protected readonly message = signal('');
  protected readonly firstName = signal('Marie');
  protected readonly lastName = signal('PELISSIER');
  protected readonly fullName = computed(() => `${this.firstName()} ${this.lastName().toUpperCase()}`);

  // der Handler des Ereignisses incremented: er erhält die ausgelösten Daten ($event)
  protected onChange(label: string, value: number): void {
    this.total.update((t) => t + 1);
    this.message.set(`${label} is now ${value}`);
  }

  protected clear(): void {
    this.firstName.set('');
    this.lastName.set('');
  }
}

// 12-projection/src/app/card/card.ts - ein wiederverwendbares TEMPLATE
// -------------------------------------------------------------------------
// Inputs übergeben DATEN; die Inhalts-PROJEKTION übergibt
// INHALT (HTML, andere Komponenten), den das Elternelement ZWISCHEN den Tags
// <app-card> und </app-card> schreibt. Die Card legt den Rahmen fest (Rand,
// Kopf- und Fußbereich), das Elternelement liefert, was hineinkommt:
//   - <ng-content />                   : der "Haupt"-Inhalt;
//   - <ng-content select="[header]" /> : nur die Elemente, die das
//                                        Attribut header haben;
//   - der Inhalt zwischen <ng-content> und </ng-content> wird angezeigt, wenn
//     das Elternelement nichts liefert (Standardinhalt).
// Es ist das clientseitige Äquivalent des Templates layout.hbs und seines {{{body}}}
// im NestJS-Kurs.
// =========================================================================

import { Component, input } from '@angular/core';

@Component({
  selector: 'app-card',
  template: `
    <div class="card mb-3" [class]="color() ? 'border-' + color() : ''">
      <div class="card-header" [class]="color() ? 'text-bg-' + color() : ''">
        <ng-content select="[header]">Untitled</ng-content>
      </div>
      <div class="card-body">
        <ng-content />

// 13-lifecycle/src/app/clock/clock.ts
// -------------------------------------------------------------------------
// Der Lebenszyklus einer Komponente:
//   1. constructor      : die Klasse wird erzeugt; die Inputs sind noch
//                         nicht verfügbar;
//   2. ngOnInit         : die Inputs haben ihren ersten Wert erhalten;
//   3. afterNextRender  : die Komponente wird ANGEZEIGT, ihre HTML-Elemente
//                         existieren in der Seite;
//   4. ngOnDestroy      : die Komponente wird gleich aus der Seite ENTFERNT.
// Eine Uhr startet beim Start einen Timer (setInterval) ... und muss ihn
// unbedingt stoppen, wenn sie verschwindet: Andernfalls würde der Timer
// sinnlos weiterlaufen (Speicherleck).
// =========================================================================

import { afterNextRender, Component, OnDestroy, OnInit, output, signal } from '@angular/core';

@Component({
  selector: 'app-clock',
  template: `<div class="display-6 font-monospace">{{ time() }}</div>`,
})
// implements OnInit, OnDestroy: TypeScript prüft, dass die Methoden existieren
export class Clock implements OnInit, OnDestroy {
  readonly log = output<string>();
  protected readonly time = signal(new Date().toLocaleTimeString());
  private timer: number | undefined;

  constructor() {
    console.log('Clock: constructor'); // das Elternelement hört noch nicht auf log
    // afterNextRender wird im Konstruktor registriert; die Funktion wird
    // einmal aufgerufen, direkt nach der ersten Anzeige der Komponente
    afterNextRender(() => this.log.emit('Clock: afterNextRender - displayed'));
  }

  ngOnInit(): void {
    this.log.emit('Clock: ngOnInit - starting the timer');
    this.timer = window.setInterval(() => this.time.set(new Date().toLocaleTimeString()), 1000);
  }

  ngOnDestroy(): void {
    window.clearInterval(this.timer);
    this.log.emit('Clock: ngOnDestroy - timer stopped');
  }
}

    • Zeilen 28–33: der Konstruktor: Die Klasse wird erzeugt. Den Eingaben wurden noch keine Werte zugewiesen, und das Elternelement hört die Ausgaben noch nicht ab (daher das [console.log] in Zeile 29 anstelle eines [emit]);
    • Zeilen 35–38: [ngOnInit()]: Die Eingaben haben ihre ersten Werte erhalten. Dies ist der übliche Ort, um Aktionen auszulösen, die von ihnen abhängen: ein Ladevorgang, ein Timer;
    • Zeile 32: [afterNextRender(function)], im Konstruktor aufgerufen, registriert eine Funktion, die einmal ausgeführt wird, unmittelbar nachdem die Komponente zum ersten Mal angezeigt wurde: Ihre HTML-Elemente existieren dann auf der Seite;
    • Zeilen 40–43: [ngOnDestroy()]: Die Komponente wird gleich von der Seite entfernt. Der Timer wird gestoppt.

    [implements OnInit, OnDestroy] (Zeile 23) ist nicht erforderlich: [Angular] ruft [ngOnInit] auf, falls die Methode existiert. [TypeScript] prüft dann jedoch, ob die Methoden korrekt geschrieben sind.

    [Angular] bietet weitere Phasen an, die seit der Einführung der Signale weniger nützlich sind: [ngOnChanges] (eine Eingabe hat sich geändert; hierfür bevorzugen wir ein [computed] oder ein [effect] auf dieser Eingabe), [ngAfterViewInit] ...     3.5. Beispiel 14: Services und Dependency Injection

    Ein Service ist eine Klasse, die Daten und Code bündelt, die von mehreren Komponenten gemeinsam genutzt werden: Serverzugriff, der angemeldete Benutzer, ein Warenkorb usw. Komponenten erzeugen keine Services (niemals [new CartService()]): Sie fordern sie von [Angular] mit der Funktion [inject()] an. Das ist Dependency Injection, der Mechanismus, den der [NestJS]-Kurs auf der Serverseite vorgestellt hat.

    Wenn eine Komponente einen Service anfordert, sucht [Angular] danach, wer ihn bereitstellt: die Komponente selbst, dann ihr Elternelement, dessen Elternelement ... bis hinauf zur Anwendung. Je nachdem, wo er bereitgestellt wird, wird ein Service daher mehr oder weniger weit geteilt. Dieses Beispiel zeigt drei Ebenen.

@Directive({
  selector: '[appLongPress]',
  host: {
    '[class.pressed]': 'pressing()',
    '(pointerdown)': 'start()',
    '(pointerup)': 'stop()',
    '(pointerleave)': 'stop()',
  },
})
export class LongPressDirective {
  readonly duration = input(1000); // in Millisekunden
  readonly appLongPress = output<void>();
  protected readonly pressing = signal(false);
  private timer: number | undefined;

  constructor() {
    inject(DestroyRef).onDestroy(() => window.clearTimeout(this.timer));
  }

  protected start(): void {
    this.pressing.set(true);
    this.timer = window.setTimeout(() => {
      this.pressing.set(false);
      this.appLongPress.emit();
    }, this.duration());
  }

  protected stop(): void {
    window.clearTimeout(this.timer);
    this.pressing.set(false);
  }
}

    // =========================================================================
// 16-confirmation/src/app/confirmation/confirmation.service.ts
// -------------------------------------------------------------------------
// "Are you sure?": window.confirm() gibt es zwar, aber es lässt sich weder
// gestalten, noch lassen sich seine Buttons übersetzen. Wir ersetzen es durch einen Bootstrap-Dialog
// (confirm-dialog.ts), der von diesem Service gesteuert wird.
//
// Der Trick: confirm(message) liefert ein PROMISE, das mit true (Ja)
// oder false (Nein) aufgelöst wird, wenn der Benutzer antwortet. Der aufrufende Code
// ist dann so einfach geschrieben wie mit window.confirm:
//
//     if (await this.confirmation.confirm('Delete this doctor?')) { ... }
//
// providedIn: 'root': nur ein Service für die ganze Anwendung, also nur ein
// Dialog, einmal in der Wurzelkomponente platziert.
// =========================================================================

import { Injectable, signal } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ConfirmationService {
  private readonly _message = signal<string | null>(null); // null: Dialog geschlossen
  readonly message = this._message.asReadonly();
  private answer: ((yes: boolean) => void) | null = null;

  confirm(text: string): Promise<boolean> {
    this.answer?.(false); // eine noch offene Frage? Sie zählt als "Nein"
    this._message.set(text);
    return new Promise<boolean>((resolve) => {
      this.answer = resolve;
    });
  }

  respond(yes: boolean): void {
    this._message.set(null);
    const r = this.answer;
    this.answer = null;
    r?.(yes);
  }
}
    • Zeile 20: ein einziger Service für die gesamte Anwendung; es gibt daher nur ein Fenster, das ein für alle Mal in der Root-Komponente platziert wird;
    • Zeile 22: die angezeigte Nachricht; [null] bedeutet, dass das Fenster geschlossen ist;
    • Zeile 23: [asReadonly()]: Komponenten können die Nachricht lesen, aber nur der Service kann sie ändern;
    • Zeile 24: die Funktion, die das aktuelle Promise auflöst, das auf eine Antwort wartet;
    • Zeilen 26–32: [confirm] öffnet das Fenster (Zeile 28) und gibt ein Promise zurück. Seine Funktion [resolve] wird beiseitegelegt (Zeile 30): Das Promise wird erst später aufgelöst, wenn der Benutzer auf eine Schaltfläche klickt;
    • Zeile 27: Wäre noch eine Frage offen, würde sie mit „Nein“ beantwortet: Ihr Promise bleibt nicht für immer unaufgelöst;
    • Zeilen 34–39: [respond] wird vom Fenster aufgerufen: Es schließt das Fenster (Zeile 35) und löst dann das Promise mit der Antwort des Benutzers auf (Zeile 38).

    Das ist der Vorteil des Promise: Der Code, der die Frage stellt, schreibt sie als Aufruf von [window.confirm()] und wartet mit [await] auf die Antwort, ohne Callback-Funktion.
