    // =========================================================================
// 01-first-application/src/app/app.ts - eine erste Komponente
// -------------------------------------------------------------------------
// Eine Angular-Komponente ist eine TypeScript-KLASSE, dekoriert mit @Component:
//   - selector    : das HTML-Tag, das sie repräsentiert (<app-root>);
//   - templateUrl : die Datei mit ihrem HTML (app.html);
//   - styleUrl    : die Datei mit ihrem CSS (app.css), das nur für
//                   diese Komponente gilt.
// Die Eigenschaften und Methoden der Klasse können im Template verwendet werden.
// =========================================================================

import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
  // "gewöhnliche" Daten: das Template kann sie anzeigen
  protected readonly title = 'Welcome to Angular';
  protected readonly languages = ['TypeScript', 'HTML', 'CSS'];

  // signal(...): ein REAKTIVER Datenwert. Sein Wert wird gelesen, indem man das Signal
  // AUFRUFT (counter()), und mit set() oder update() geändert. Angular
  // aktualisiert dann AUTOMATISCH die Stellen der Seite, die ihn anzeigen.
  protected readonly counter = signal(0);
  protected readonly time = signal(new Date().toLocaleTimeString());

  // eine Methode, die vom Button des Templates aufgerufen wird
  protected increment(): void {
    this.counter.update((n) => n + 1);
    this.time.set(new Date().toLocaleTimeString());
  }
}
    • Zeilen 14–18: Der Dekorator [@Component] macht die Klasse zu einer Komponente:
      • Zeile 15: ihr Selektor, das Tag, das sie in einem Template oder in [index.html] repräsentiert;
      • Zeile 16: ihre Template-Datei, HTML;
      • Zeile 17: ihre Stildatei, CSS;
    • Zeilen 21–22: zwei gewöhnliche Datenelemente. [protected]: Sie werden nur von der Komponente und ihrem Template verwendet; [readonly]: Sie werden nicht ersetzt;
    • Zeile 27: [signal(0)] erzeugt einen reaktiven Datenwert mit dem Anfangswert 0. Ein Signal ist eine Funktion: Seinen Wert lesen Sie, indem Sie es aufrufen ([this.counter()]), und ändern ihn mit [set(value)] oder [update(function)]. Wenn es sich ändert, aktualisiert [Angular] alle Stellen der Seite, die es anzeigen;
    • Zeilen 31–34: eine Methode, die zwei Signals ändert. Sie ändert die Seite nicht: [Angular] aktualisiert sie. [update((n) => n + 1)] berechnet den neuen Wert anhand des alten.

    // =========================================================================
// 02-signals/src/app/app.ts - signal, computed, effect, linkedSignal
// -------------------------------------------------------------------------
// Ein kleiner Preisrechner mit Mehrwertsteuer:
//   - signal       : ein reaktiver Wert, der durch Aufruf gelesen und
//                    mit set() / update() geändert wird;
//   - computed     : ein aus anderen Signalen BERECHNETER Wert, der
//                    automatisch neu berechnet wird, wenn sich diese ändern;
//   - effect       : Code, der bei jeder Änderung der Signale ausgeführt wird,
//                    die er liest (ein "Seiteneffekt");
//   - linkedSignal : ein beschreibbares Signal, das aber auf einen berechneten Wert
//                    ZURÜCKGESETZT wird, wenn sich ein anderes Signal ändert.
// =========================================================================

import { Component, computed, effect, linkedSignal, signal, untracked } from '@angular/core';
// FormsModule stellt [(ngModel)] bereit, das ein Feld mit einem Signal verknüpft (siehe Beispiel 05)
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly label = signal('Pen');
  protected readonly priceExclVat = signal(2.5);
  protected readonly vatRate = signal(20); // in %

  // linkedSignal: die Menge springt bei jedem Wechsel des Artikels auf 1 zurück,
  // der Benutzer kann sie zwischenzeitlich aber ändern
  protected readonly quantity = linkedSignal({ source: this.label, computation: () => 1 });

  // computed: wird nie direkt zugewiesen, sondern immer neu berechnet. Angular weiß,
  // von welchen Signalen es abhängt (die es LIEST): es berechnet es neu, wenn sich eines
  // davon ändert, und nur dann
  protected readonly totalExclVat = computed(() => this.quantity() * this.priceExclVat());
  protected readonly vatAmount = computed(() => (this.totalExclVat() * this.vatRate()) / 100);
  protected readonly totalInclVat = computed(() => this.totalExclVat() + this.vatAmount());

  // ein Formatierer für Beträge in Euro im englischen Stil (€1,234.50)
  private readonly euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });

  // der Verlauf der Gesamtbeträge inkl. MwSt., gefüllt vom folgenden Effect
  protected readonly history = signal<string[]>([]);
  private oldTotal: number | null = null;

  constructor() {
    // effect: wird zuerst einmal ausgeführt, dann bei jeder Änderung von
    // totalInclVat (dem einzigen Signal, das er liest)
    effect(() => {
      const total = this.totalInclVat();
      document.title = `Incl. VAT: ${this.format(total)}`;
      if (this.oldTotal !== null) {
        const line = `${this.format(this.oldTotal)} -> ${this.format(total)}`;
        // untracked: history ändern, OHNE es zu einer Abhängigkeit des Effects zu machen
        untracked(() => this.history.update((h) => [line, ...h].slice(0, 5)));
      }
      this.oldTotal = total;
    });
  }

  protected format(amount: number): string {
    return this.euros.format(amount);
  }

  protected reset(): void {
    this.label.set('Pen');

//   - (click)="f(3, $event)"   ruft f mit Argumenten auf; $event ist
//                              das native Ereignis des Browsers;
//   - (keyup.enter), (keydown.escape): nur für diese Taste;

//     das HTML-Element selbst bezeichnet (name.value, name.focus()...).

import { Component, signal } from '@angular/core';

interface Task {
  id: number;
  label: string;
  done: boolean;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .area {
      background: #e9f2ff;
      height: 80px;
      cursor: crosshair;
    }
  `,
})
export class App {
  protected readonly tasks = signal<Task[]>([
    { id: 1, label: 'Install Node.js', done: true },
    { id: 2, label: 'Read the chapter on events', done: false },
  ]);
  protected readonly log = signal<string[]>([]);
  private nextId = 3;

  private trace(message: string): void {
    this.log.update((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // aufgerufen von (submit): $event.preventDefault() verhindert, dass der Browser
  // das Formular an den Server sendet (und die Seite neu lädt)
  protected add(event: SubmitEvent, field: HTMLInputElement): void {
    event.preventDefault();
    const label = field.value.trim();
    if (label === '') {
      return;
    }
    // ein Signal mit einem Array: wir ERSETZEN es durch ein neues Array
    this.tasks.update((t) => [...t, { id: this.nextId++, label, done: false }]);
    this.trace(`added "${label}"`);
    field.value = '';
    field.focus();
  }

  protected toggle(task: Task, done: boolean): void {
    this.tasks.update((t) => t.map((x) => (x.id === task.id ? { ...x, done } : x)));
    this.trace(`"${task.label}": ${done ? 'done' : 'to do'}`);
  }

  protected remove(task: Task): void {
    this.tasks.update((t) => t.filter((x) => x.id !== task.id));
    this.trace(`deleted "${task.label}"`);
  }

  // $event: das native Ereignis (hier ein MouseEvent), mit seinen Koordinaten
  protected position(event: MouseEvent): void {
    this.trace(`click at (${event.offsetX}, ${event.offsetY})`);
  }
}

// -------------------------------------------------------------------------
// Die moderne Art (Angular 22), ein Formular zu schreiben:
//   1. das MODELL: ein Signal, das ein Objekt mit einer Eigenschaft pro
//      Feld enthält (das Formular liest und schreibt in diesem Signal);
//   2. form(model, schema): erstellt einen FELDBAUM (FieldTree), der
//      dieselbe Form wie das Modell hat: f.lastName, f.age, f.newsletter... Das
//      Schema (optional) beschreibt die Regeln der Felder (siehe Beispiel 07);
//   3. [formField]="f.lastName" verknüpft ein HTML-Feld mit einem Feld des Baums.
// Jedes Feld hat einen ZUSTAND, den man durch Aufruf erhält: f.lastName() liefert seinen Wert
// (value()), ob es berührt wurde (touched()), geändert wurde (dirty()), ob es
// deaktiviert ist (disabled())...
// =========================================================================

import { Component, signal } from '@angular/core';
import { JsonPipe } from '@angular/common';
import { disabled, form, FormField } from '@angular/forms/signals';

// die Form der Formulardaten
interface Profile {
  title: string;
  firstName: string;
  lastName: string;
  age: number;
  newsletter: boolean;
  email: string;
  country: string;
  birthDate: string;
  presentation: string;
}

const INITIAL: Profile = {
  title: 'Mme',
  firstName: 'Marie',
  lastName: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  country: 'fr',
  birthDate: '',
  presentation: '',
};

@Component({
  selector: 'app-root',
  imports: [FormField, JsonPipe],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = ['Mr', 'Mme', 'Melle'];

  // 1. das Modell
  protected readonly model = signal<Profile>({ ...INITIAL });

  // 2. der Feldbaum; eine Regel: die E-Mail kann nur geändert werden, wenn das Kästchen
  // "newsletter" angehakt ist (valueOf liest den Wert eines ANDEREN Felds)
  protected readonly f = form(this.model, (p) => {
    disabled(p.email, ({ valueOf }) => !valueOf(p.newsletter));
  });

  // die Felder, deren Zustand in der Tabelle rechts angezeigt wird
  protected readonly fields = ['firstName', 'lastName', 'age', 'email'] as const;

  protected reset(): void {
    // das Modell kehrt zu seinem Anfangswert zurück und der Zustand der Felder wird zurückgesetzt
    this.model.set({ ...INITIAL });
    this.f().reset();
  }

  protected upperCase(): void {
    // das MODELL per Code ändern: die Felder folgen
    this.model.update((m) => ({ ...m, lastName: m.lastName.toUpperCase(), firstName: m.firstName.toUpperCase() }));
  }
}

// -------------------------------------------------------------------------
// Die Regeln der Felder stehen im SCHEMA des Formulars (zweites
// Argument von form()): required, maxLength, minLength, pattern... und
// validate() für eine "selbstgeschriebene" Regel. Jedes Feld kennt dann seine
// Fehler (f.lastName().errors()), die bei jedem Tastendruck neu berechnet werden.
// Sie werden nicht sofort angezeigt (ein komplett rotes leeres Formular ist
// unangenehm): nur für ein bereits verlassenes Feld ("touched"). submit()
// markiert alle Felder als touched und führt die Aktion nur aus, wenn das
// Formular gültig ist.
//
// Dieselben Regeln wie beim RdvMedecins-Server (Formular "Konto erstellen").
// ACHTUNG: Diese Validierung ist nur eine ERLEICHTERUNG für den Benutzer (eine
// sofortige Rückmeldung, ohne Roundtrip zum Server). Sie schützt
// nichts: Jeder kann dem Server eine von Hand gebaute Anfrage
// senden (curl...). Der Server validiert deshalb IMMER alles erneut (siehe Beispiel 25).
// =========================================================================

import { Component, signal } from '@angular/core';
import { form, FormField, maxLength, minLength, pattern, required, submit, validate } from '@angular/forms/signals';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

const TITLES = ['Mr', 'Mme', 'Melle'];
const LOGIN_FORMAT = /^[a-z0-9._-]{1,30}$/;
const PASSWORD_MIN = 6;

@Component({
  selector: 'app-root',
  imports: [FormField],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = TITLES;
  protected readonly model = signal<Account>({
    title: 'Mr',
    firstName: '',
    lastName: '',
    login: '',
    password: '',
    confirmation: '',
  });

  // das Schema: p bezeichnet die PFADE der Felder (p.lastName, p.login...)
  protected readonly f = form(this.model, (p) => {
    validate(p.title, ({ value }) =>
      TITLES.includes(value()) ? null : { kind: 'title', message: 'Choose a title from the list.' },
    );
    required(p.firstName, { message: 'The first name is required.' });
    maxLength(p.firstName, 20, { message: 'The first name must not exceed 20 characters.' });
    required(p.lastName, { message: 'The last name is required.' });
    maxLength(p.lastName, 20, { message: 'The last name must not exceed 20 characters.' });
    required(p.login, { message: 'The login is required.' });
    pattern(p.login, LOGIN_FORMAT, { message: 'Unaccented lower-case letters, digits, dot, hyphen, underscore.' });
    required(p.password, { message: 'The password is required.' });
    minLength(p.password, PASSWORD_MIN, { message: `At least ${PASSWORD_MIN} characters.` });
    // eine "selbstgeschriebene" Regel, die ZWEI Felder vergleicht (valueOf liest das andere)
    validate(p.confirmation, ({ value, valueOf }) =>
      value() === valueOf(p.password)
        ? null
        : { kind: 'confirmation', message: 'The confirmation does not match the password.' },
    );
  });

  protected readonly submitAttempted = signal(false);
  protected readonly success = signal(false);

  // (submit) des Formulars
  protected async send(event: SubmitEvent): Promise<void> {
    event.preventDefault();
    this.submitAttempted.set(true);
    // submit() markiert alle Felder als touched (ihre Fehler
    // erscheinen) und führt die Aktion nur aus, wenn alles gültig ist
    const valid = await submit(this.f, async () => {
      // hier würden wir das Modell an den Server senden (siehe Beispiel 25)
      return null; // null: kein Fehler
    });
    this.success.set(valid);
  }
}

import { Component, inject, signal } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

const TITLES = ['Mr', 'Mme', 'Melle'];

// ein "selbstgeschriebener" Validator gilt für die GRUPPE: er vergleicht zwei ihrer Felder.
// Er liefert null (kein Fehler) oder ein Objekt, das den Fehler beschreibt.
function sameConfirmation(group: AbstractControl): ValidationErrors | null {
  const password = group.get('password')?.value;
  const confirmation = group.get('confirmation')?.value;
  return password === confirmation ? null : { confirmation: true };
}

@Component({
  selector: 'app-root',
  imports: [ReactiveFormsModule],
  templateUrl: './app.html',
})

  protected readonly titles = TITLES;
  private readonly fb = inject(FormBuilder).nonNullable;

  // die Gruppe: { name: [Anfangswert, Validatoren] }
  protected readonly form = this.fb.group(
    {
      title: ['Mr', [Validators.required]],
      firstName: ['', [Validators.required, Validators.maxLength(20)]],
      lastName: ['', [Validators.required, Validators.maxLength(20)]],
      login: ['', [Validators.required, Validators.pattern(/^[a-z0-9._-]{1,30}$/)]],
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmation: [''],
    },
    { validators: sameConfirmation },
  );

  protected readonly success = signal(false);
  protected readonly value = signal('');

  // ein Feld wird als fehlerhaft markiert, wenn es touched und ungültig ist
  protected inError(name: string): boolean {
    const control = this.form.get(name);
    return !!control && control.touched && control.invalid;
  }

  // (ngSubmit): alle Felder werden "touched", ihre Fehler erscheinen
  protected send(): void {
    this.form.markAllAsTouched();
    this.success.set(this.form.valid);
    // getRawValue(): der Wert der gesamten Gruppe, typisiert
    this.value.set(JSON.stringify(this.form.getRawValue()));
  }

// 09-pipes/src/app/pipes/time-range.pipe.ts - eine "selbstgeschriebene" Pipe
// -------------------------------------------------------------------------
// Eine Pipe ist eine mit @Pipe dekorierte Klasse, deren Methode transform()
// den Wert VOR dem senkrechten Strich (und gegebenenfalls
// Parameter) erhält und zurückgibt, was angezeigt werden soll:
//     {{ slot | timeRange }}           -> "8:00 - 8:20"
//     {{ slot | timeRange: 'start' }}  -> "8:00"
// Eine Pipe ist standardmäßig "pure": Angular ruft transform() nur erneut auf, wenn sich
// der Wert oder die Parameter GEÄNDERT haben (gleiches Objekt = gleiches Ergebnis).
// =========================================================================

import { Pipe, PipeTransform } from '@angular/core';

export interface Slot {
  startHour: number;
  startMinute: number;
  endHour: number;
  endMinute: number;
}

// "8:05" statt "8:5"
function time(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

@Pipe({ name: 'timeRange' })
export class TimeRangePipe implements PipeTransform {
  transform(c: Slot, part: 'all' | 'start' | 'end' = 'all'): string {
    switch (part) {
      case 'start':
        return time(c.startHour, c.startMinute);
      case 'end':
        return time(c.endHour, c.endMinute);
      default:
        return `${time(c.startHour, c.startMinute)} - ${time(c.endHour, c.endMinute)}`;
    }
  }
}
    • Zeile 27: der Name der Pipe, der in den Templates geschrieben wird;
    • Zeile 29: [transform(value, part)] gibt zurück, was angezeigt werden soll. Der Parameter ist optional (Standardwert ['all']): [{{ slot | timeRange }}] zeigt [8:00 - 8:20] an, [{{ slot | timeRange: 'start' }}] zeigt [8:00] an.

    Eine Pipe ist standardmäßig pure: [Angular] ruft [transform] nur auf, wenn sich der Wert oder einer der Parameter geändert hat (ein anderes Objekt, ein anderer String). Das macht sie effizienter als eine Methode der Komponente, die im Template aufgerufen wird (wie [format(...)] in Beispiel 02) und bei jeder Prüfung der Komponente ausgeführt wird.
