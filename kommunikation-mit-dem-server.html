    8. Kommunikation mit dem Server

    Die Beispiele dieses Kapitels fragen den JSON-Server [RdvMedecins] ab: Er muss laufen ([npm run start:dev] im Ordner [rdvmedecins-nestjs-json]) auf Port 8080, gleichzeitig mit dem [Angular]-Entwicklungsserver auf Port 4200.

    8.1. Beispiel 22: Eine erste HTTP-Anfrage

    8.1.1. Der Proxy des Entwicklungsservers

    Die Beispielseite stammt von http://localhost:4200 (dem [Angular]-Entwicklungsserver, gestartet mit [ng serve]); die API befindet sich unter http://localhost:8080. Für einen Browser sind dies zwei verschiedene Ursprünge (der Port ist Teil des Ursprungs). Aus Sicherheitsgründen erlaubt ein Browser einer Seite jedoch nicht, Antworten eines anderen Ursprungs zu lesen, es sei denn, der Server erlaubt es ausdrücklich (der Mechanismus, der als CORS bekannt ist). Und eine Anfrage an einen anderen Ursprung überträgt keine Cookies (und damit auch nicht das Token), es sei denn, der Code verlangt es ([withCredentials: true]) UND der Server akzeptiert es, wiederum über CORS.

    Anstatt den Server für andere Ursprünge zu öffnen, leiten wir die API-Anfragen über den [Angular]-Entwicklungsserver. Das ist der Proxy , der in der Workspace-Datei [proxy.conf.json] deklariert ist:

// 22-httpclient/src/app/app.ts - eine erste HTTP-Anfrage an den Server
// -------------------------------------------------------------------------
// Der JSON-Server von RdvMedecins muss laufen (Port 8080). Wir fordern von ihm
// ein Captcha an: GET /api/captcha -> { "id": "...", "svg": "<svg ...>" }.
//
//   - HttpClient: der HTTP-Client von Angular (standardmäßig für die gesamte
//     Anwendung bereitgestellt). http.get<T>(url) liefert ein OBSERVABLE (siehe Beispiel
//     19): Die Anfrage wird erst gesendet, wenn wir es ABONNIEREN (subscribe);
//     die JSON-Antwort kommt bereits in ein Objekt des angegebenen Typs T umgewandelt an;
//   - die Adresse ist RELATIV (/api/captcha): die Anfrage geht an den
//     Server der Seite, den von Angular (Port 4200), der sie dank seines Proxys (siehe proxy.conf.json) an den
//     JSON-Server (Port 8080) weiterleitet;
//   - drei anzuzeigende Zustände: Laden, Erfolg, Fehler.
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { DomSanitizer, SafeHtml } from '@angular/platform-browser';

// was GET /api/captcha zurückgibt
interface Captcha {
  id: string;
  svg: string;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .captcha {
      display: inline-block;
      border: 1px solid #dee2e6;
      border-radius: 0.375rem;
    }
  `,
})
export class App {
  private readonly http = inject(HttpClient);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly captcha = signal<Captcha | null>(null);
  protected readonly loading = signal(false);
  protected readonly error = signal<string | null>(null);
  protected readonly requestCount = signal(0);

  // [innerHTML] fügt einen String als HTML ein. Aus Sicherheitsgründen BEREINIGT
  // Angular dieses HTML ("Sanitization"): Es entfernt die Skripte ... und die
  // <svg>-Tags. bypassSecurityTrustHtml sagt Angular: "Dieses HTML ist
  // sicher, nicht bereinigen". NUR für sichere Inhalte verwenden: Dieses SVG
  // wird von UNSEREM Server erzeugt. Bei Benutzereingaben wäre das
  // ein Einfallstor für Script-Injection ({{ }} dagegen maskiert alles).
  protected readonly svg = computed<SafeHtml | null>(() => {
    const c = this.captcha();
    return c ? this.sanitizer.bypassSecurityTrustHtml(c.svg) : null;
  });

  constructor() {
    this.load(); // beim Start laden wir ein Captcha
  }

  protected load(): void {
    this.loading.set(true);
    this.error.set(null);
    this.http.get<Captcha>('/api/captcha').subscribe({
      // next: die Antwort (Code 2xx), bereits aus JSON umgewandelt
      next: (captcha) => {
        this.captcha.set(captcha);
        this.requestCount.update((n) => n + 1);
        this.loading.set(false);
      },
      // error: eine Antwort 4xx / 5xx oder gar keine Antwort (Status 0)
      error: (e: HttpErrorResponse) => {
        this.error.set(`the server answered ${e.status} ${e.statusText}`);
        this.captcha.set(null);
        this.loading.set(false);
      },
    });
  }
}

    // =========================================================================
// 23-login/src/app/app.ts - Anmeldung am RdvMedecins-Server
// -------------------------------------------------------------------------
//   1. beim Start: GET /api/auth/me - sind wir schon angemeldet?
//      (nach einem F5 hat der Client alles vergessen; der Server liest das Cookie);
//   2. Login: POST /api/auth/login, JSON-Body { login, password,
//      captchaId, captcha }. Bei Erfolg setzt der Server das
//      httpOnly-Cookie des JWT-Tokens: Unser Code sieht es NIEMALS (probieren Sie
//      document.cookie in der Konsole: Es ist nicht da);
//   3. geschützte Anfrage: GET /api/doctors. Der Browser hängt das Cookie
//      von selbst an (gleicher Origin);
//   4. Logout: POST /api/auth/logout - der Server löscht das Cookie.
//
// firstValueFrom(observable): ein PROMISE auf den ersten Wert des Streams.
// Anfragenketten lassen sich dann mit async / await schreiben,
// lesbarer als eine Kaskade von subscribe. Eine Fehlerantwort lehnt
// das Promise mit einer HttpErrorResponse ab: ein einfaches try / catch.
//
// Ein Cookie gehört zu einem Rechner (localhost) und zu einem Pfad (/api): Es
// ist daher ALLEN Beispielen gemeinsam. Sobald Sie hier angemeldet sind, sind es die Beispiele
// 24 und 25 ebenfalls.
// =========================================================================

import { Component, computed, inject, signal } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { DomSanitizer } from '@angular/platform-browser';
import { FormsModule } from '@angular/forms';
import { firstValueFrom } from 'rxjs';

interface User {
  login: string;
  name: string;
  role: 'ADMIN' | 'DOCTOR' | 'USER';
}
interface Doctor {
  id: number;
  title: string;
  firstName: string;
  lastName: string;
}
interface Captcha {
  id: string;
  svg: string;
}
// die Form ALLER Fehler der API
interface ApiError {
  statusCode: number;
  key: string;
  params?: Record<string, string>;
}

// die wenigen Meldungen, die dieses Beispiel braucht (Beispiel 25 verwendet ngx-translate)
const MESSAGES: Record<string, string> = {
  'LOGIN.ERROR': 'Incorrect login or password.',
  'ERRORS.WRONG_CAPTCHA': 'The characters you typed do not match the image.',
  'ERRORS.TOO_MANY_ATTEMPTS': 'Too many failed attempts. Try again later.',
};

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  private readonly http = inject(HttpClient);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly user = signal<User | null>(null);
  protected readonly doctors = signal<Doctor[]>([]);
  protected readonly captcha = signal<Captcha | null>(null);
  protected readonly svg = computed(() => this.sanitizer.bypassSecurityTrustHtml(this.captcha()?.svg ?? ''));
  protected readonly login = signal('admin');
  protected readonly password = signal('');
  protected readonly captchaAnswer = signal('');
  protected readonly error = signal<string | null>(null);
  protected readonly ready = signal(false); // ist die anfängliche Prüfung abgeschlossen?

  constructor() {
    this.checkSession().catch(() => {
      this.error.set('Is the RdvMedecins JSON server running (port 8080)?');
      this.ready.set(true);
    });
  }

  protected async loadCaptcha(): Promise<void> {
    this.captcha.set(await firstValueFrom(this.http.get<Captcha>('/api/captcha')));
    this.captchaAnswer.set('');
  }

  // 1. wer ist angemeldet? 200 -> der Benutzer; 401 -> niemand
  private async checkSession(): Promise<void> {
    try {
      this.user.set(await firstValueFrom(this.http.get<User>('/api/auth/me')));
      await this.loadDoctors();
    } catch (e) {
      if (!(e instanceof HttpErrorResponse) || e.status !== 401) {
        throw e; // kein 401: der Server antwortet nicht
      }
      await this.loadCaptcha();
    }
    this.ready.set(true);
  }

  // 2. der Login; post(url, body): der Body wird als JSON gesendet
  protected async logIn(): Promise<void> {
    this.error.set(null);
    try {
      const u = await firstValueFrom(
        this.http.post<User>('/api/auth/login', {
          login: this.login(),
          password: this.password(),
          captchaId: this.captcha()?.id,
          captcha: this.captchaAnswer(),
        }),
      );
      this.user.set(u);
      this.password.set('');
      await this.loadDoctors();
    } catch (e) {
      // e.error: der JSON-Body der Fehlerantwort
      const apiError = (e as HttpErrorResponse).error as ApiError | null;
      const key = apiError?.key ?? 'ERRORS.SERVER_UNREACHABLE';
      this.error.set(`${(e as HttpErrorResponse).status} - ${MESSAGES[key] ?? key}`);
      await this.loadCaptcha(); // jedes Bild wird nur einmal verwendet
    }
  }

  // 3. eine geschützte Anfrage
  private async loadDoctors(): Promise<void> {
    this.doctors.set(await firstValueFrom(this.http.get<Doctor[]>('/api/doctors')));
  }

  // 4. der Logout
  protected async logOut(): Promise<void> {
    await firstValueFrom(this.http.post('/api/auth/logout', {}));
    this.user.set(null);
    this.doctors.set([]);
    await this.loadCaptcha();
  }
}

// common/api-error.ts - EINE Klasse für alle Fehler der API
// -------------------------------------------------------------------------
// Alle Fehlerantworten des RdvMedecins-Servers haben dieselbe Form:
//   { "statusCode": 409, "key": "ERRORS.LOGIN_TAKEN", "params": {...}, "fields": {...} }
// HttpClient meldet sie mit einer HttpErrorResponse, deren Eigenschaft error
// diesen JSON-Body enthält ... oder etwas anderes: null (keine Antwort), Text
// (eine Fehlerseite des Proxys, wenn der JSON-Server gestoppt ist). Wir
// wandeln all das in EINE Klasse um, ApiError, die den vom Server gesendeten Übersetzungsschlüssel
// trägt (und gegebenenfalls dessen Parameter und
// den Fehler jedes Felds).
// =========================================================================

import { HttpErrorResponse } from '@angular/common/http';

export class ApiError extends Error {
  constructor(
    public readonly statusCode: number, // 0: der Server hat nicht geantwortet
    public readonly key: string, // z. B.: 'ERRORS.LOGIN_TAKEN'
    public readonly params: Record<string, string> = {},
    public readonly fields: Record<string, string> = {},
  ) {
    super(key);
    this.name = 'ApiError';
  }
}

// jeder Fehler -> ein ApiError
export function toApiError(e: unknown): ApiError {
  if (e instanceof ApiError) {
    return e;
  }
  if (!(e instanceof HttpErrorResponse)) {
    return new ApiError(0, 'ERRORS.SERVER_ERROR');
  }
  // der JSON-Body des Fehlers, falls vorhanden und von der richtigen Form
  const body = (typeof e.error === 'object' && e.error !== null ? e.error : {}) as {
    key?: string;
    params?: Record<string, string>;
    fields?: Record<string, string>;
  };
  // kein key: nicht unser Server hat geantwortet (Proxy, Netzwerk)
  const key = body.key ?? (e.status === 0 || e.status >= 500 ? 'ERRORS.SERVER_UNREACHABLE' : 'ERRORS.SERVER_ERROR');
  return new ApiError(e.status, key, body.params, body.fields);
}

function tomorrow(): string {
  const d = new Date(Date.now() + 24 * 3600 * 1000);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

@Component({
  selector: 'app-root',
  imports: [FormsModule, LoginBar],
  templateUrl: './app.html',

  protected readonly clients = signal<Person[]>([]);
  protected readonly doctorId = signal<number | null>(null);
  protected readonly day = signal(tomorrow());
  protected readonly clientId = signal<number | null>(null);

      this.day();
      this.message.set(null);

// Die Verwaltung der Kunden (Admin-Konto). Dieses Formular
// führt ABSICHTLICH keine Validierung im Browser durch: Sein Schema ist leer.
// Alles wird an den Server gesendet, der im Fall einer Ablehnung mit einem JSON-Fehler
// antwortet (vom Interceptor in einen ApiError umgewandelt):
//
//   400 { key: 'ERRORS.INVALID_FORM', fields: { lastName: 'ERRORS.LASTNAME_REQUIRED' } }
//   409 { key: 'ERRORS.PATIENT_ALREADY_EXISTS', fields: { firstName: ..., lastName: ... } }
//   409 { key: 'ERRORS.CONCURRENT_UPDATE' }         (optimistisches Sperren)
//
// submit(form, action): die Aktion sendet das Modell an den Server. Liefert sie
// FEHLER (ein Array) zurück, hängen die Signal-Formulare diese
// an die bezeichneten Felder (fieldTree): Sie erscheinen in f.lastName().errors(),
// wie die Fehler eines Validators. Ihre Meldung ist hier der vom
// Server gesendete KEY, den das Template übersetzt (ngx-translate). Probieren Sie: einen leeren
// Nachnamen, einen Titel "Dr", einen Namensvetter (Alice DURAND), einen bereits vergebenen Login
// (martin), denselben Datensatz in zwei Tabs geändert...

import { Component, computed, inject, signal } from '@angular/core';
import { FormField, form, submit, ValidationError } from '@angular/forms/signals';
import { TranslatePipe, TranslateService } from '@ngx-translate/core';
import { ApiService } from '../../../common/api.service';
import { LoginBar } from '../../../common/login-bar';
import { ApiError } from '../../../common/api-error';
import { Person, User } from '../../../common/types';
// der Bestätigungsdialog aus Beispiel 16, unverändert wiederverwendet
import { ConfirmationService } from '../../../16-confirmation/src/app/confirmation/confirmation.service';
import { ConfirmDialog } from '../../../16-confirmation/src/app/confirmation/confirm-dialog';

interface ClientInput {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  version: number;
}

const EMPTY: ClientInput = { title: 'Mr', firstName: '', lastName: '', login: '', password: '', version: 0 };

@Component({
  selector: 'app-root',
  imports: [FormField, TranslatePipe, LoginBar, ConfirmDialog],
  templateUrl: './app.html',
})
export class App {
  private readonly api = inject(ApiService);
  private readonly confirmation = inject(ConfirmationService);
  protected readonly translate = inject(TranslateService);

  protected readonly user = signal<User | null>(null);
  protected readonly clients = signal<Person[]>([]);
  protected readonly isAdmin = computed(() => this.user()?.role === 'ADMIN');

  // das Formular: editedId === null -> Anlegen, andernfalls Aktualisieren
  protected readonly editedId = signal<number | null>(null);
  protected readonly model = signal<ClientInput>({ ...EMPTY });
  protected readonly f = form(this.model); // kein Validator: der Server validiert
  // die allgemeine Meldung des letzten Fehlers und die eines Erfolgs
  protected readonly pageError = signal<ApiError | null>(null);
  protected readonly success = signal<{ key: string; params: Record<string, string> } | null>(null);

  protected async onUserChange(u: User | null): Promise<void> {
    this.user.set(u);
    this.clients.set([]);
    if (u?.role === 'ADMIN') {
      await this.reload();
    }
  }

  private async reload(): Promise<void> {
    this.clients.set(await this.api.get<Person[]>('/clients'));
  }

  protected newForm(): void {
    this.editedId.set(null);
    this.model.set({ ...EMPTY });
    this.f().reset(); // der Zustand der Felder (und ihre Fehler) beginnt bei null
    this.clearMessages();
  }

  protected edit(c: Person): void {
    this.editedId.set(c.id);
    // die zusammen mit dem Datensatz gelesene VERSION geht mit der Aktualisierung zurück: Hat jemand
    // den Datensatz inzwischen geändert, antwortet der Server mit 409 CONCURRENT_UPDATE
    this.model.set({ title: c.title, firstName: c.firstName, lastName: c.lastName, login: c.login ?? '', password: '', version: c.version });
    this.f().reset();
    this.clearMessages();
  }

  private clearMessages(): void {
    this.pageError.set(null);
    this.success.set(null);
  }

  // der Fehler JEDES Felds, vom Server zurückgegeben -> ein Fehler des Formulars
  private fieldErrors(e: ApiError): ValidationError.WithOptionalFieldTree[] {
    const fields = Object.keys(this.model()) as (keyof ClientInput)[];
    return Object.entries(e.fields)
      .filter(([field]) => fields.includes(field as keyof ClientInput))
      .map(([field, key]) => ({ kind: 'server', message: key, fieldTree: this.f[field as keyof ClientInput] }));
  }

  protected async save(event: SubmitEvent): Promise<void> {
    event.preventDefault();
    this.clearMessages();
    const id = this.editedId();
    await submit(this.f, async () => {
      try {
        const body = this.model();
        const client =
          id === null
            ? await this.api.post<Person>('/clients', body)
            : await this.api.put<Person>(`/clients/${id}`, body);
        this.newForm(); // Formular geleert, Meldungen gelöscht...
        // ... außer dieser
        this.success.set({
          key: id === null ? 'FORM.ADDED' : 'FORM.UPDATED',
          params: { name: `${client.firstName} ${client.lastName}`, login: client.login ?? '' },
        });
        await this.reload();
        return null;
      } catch (e) {
        if (!(e instanceof ApiError)) {
          throw e;
        }
        this.pageError.set(e); // die allgemeine Meldung
        return this.fieldErrors(e); // der Fehler jedes Felds
      }
    });
  }

  protected async remove(c: Person): Promise<void> {
    const name = `${c.firstName} ${c.lastName}`;
    if (!(await this.confirmation.confirm(this.translate.instant('FORM.DELETE_CONFIRM', { name })))) {
      return;
    }
    this.clearMessages();
    try {
      await this.api.delete(`/clients/${c.id}`);
      this.success.set({ key: 'FORM.DELETED', params: { name } });
    } catch (e) {
      this.pageError.set(e instanceof ApiError ? e : null);
    }
    await this.reload();
  }
}
